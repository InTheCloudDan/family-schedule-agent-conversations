+def _external_calendar_accounts_for_event_sync_state(
+    event: Event, requester: User
+) -> Any:
+    """Return tenant-scoped external calendars that may store sync state.
+
+    External provider calendar IDs are not globally unique in ClanCal: unrelated
+    families can import calendars with the same provider calendar_id. Sync
+    suppression/modification rows must therefore only be written for the
+    requester's account and active members of the event's family.
+    """
+    calendar_filters = Q(user=requester)
+    if event.family_group_id:
+        calendar_filters |= Q(
+            user__family_memberships__family_id=event.family_group_id,
+            user__family_memberships__is_active=True,
+        )
+
+    calendars_qs = ExternalCalendarAccount.objects.filter(
+        calendar_id=event.calendar_id
+    ).filter(calendar_filters)
+    if event.external_source:
+        calendars_qs = calendars_qs.filter(provider=event.external_source)
+
+    return calendars_qs.distinct()

 def _get_weekend_summary_signed_cookie(request: Any, summary: Any) -> Any:
     """Return a redeemed weekend summary token from a signed browser cookie."""
     try:
         return request.get_signed_cookie(
             summary.get_access_cookie_name(),
             default=None,
             salt=summary.ACCESS_COOKIE_SALT,
         )
     except BadSignature:
         logger.warning(
             "Invalid signed weekend summary cookie for summary %s", summary.id
         )
         return None
